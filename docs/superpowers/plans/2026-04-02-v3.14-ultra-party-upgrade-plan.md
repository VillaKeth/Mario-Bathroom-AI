    if self._memorial_active:
        if DEBUG_CLIENT:
            print("[DEBUG_CLIENT] Suppressed idle text during memorial")
        return
    
    text = data.get("text", "")
    emotion = data.get("emotion", "neutral")
    
    # Set the text on display
    self.display.set_speech_text(text, emotion=emotion)
    self._speaking = True
    
    # Cancel any existing clear timer
    if hasattr(self, '_bubble_clear_timer') and self._bubble_clear_timer:
        self._bubble_clear_timer.cancel()
    
    # Start a polling loop that keeps bubble until audio finishes
    def _wait_for_audio_complete():
        import time
        # Wait for audio to start playing (up to 2s)
        for _ in range(20):
            if self.audio_playback.is_playing:
                break
            time.sleep(0.1)
        # Wait for audio to finish
        while self.audio_playback.is_playing:
            time.sleep(0.1)
        # 500ms grace period after audio ends
        time.sleep(0.5)
        # Clear bubble
        self.display.clear_speech_text()
        self._speaking = False
    
    self._bubble_clear_timer = threading.Thread(
        target=_wait_for_audio_complete, daemon=True
    )
    self._bubble_clear_timer.start()

# client/closed_captions.py
"""Always-on subtitle bar at bottom of screen for Mario's speech."""
import pygame

DEBUG_CAPTIONS = False

class ClosedCaptions:
    def __init__(self, screen_width, screen_height):
        self._screen_w = screen_width
        self._screen_h = screen_height
        self._text = ""
        self._font = pygame.font.Font(None, 24)
        self._bg_color = (0, 0, 0, 180)  # semi-transparent black
        self._text_color = (255, 255, 255)
        self._padding = 10
        self._max_lines = 3
        if DEBUG_CAPTIONS:
            print("[DEBUG_CAPTIONS] ClosedCaptions: initialized")
    
    def set_text(self, text):
        if DEBUG_CAPTIONS:
            print(f"[DEBUG_CAPTIONS] set_text: {text[:50]}...")
        self._text = text
    
    def clear(self):
        self._text = ""
    
    def draw(self, surface):
        if not self._text:
            return
        
        # Word-wrap text
        words = self._text.split()
        lines = []
        current = ""
        max_w = self._screen_w - 2 * self._padding
        for word in words:
            test = f"{current} {word}".strip()
            if self._font.size(test)[0] <= max_w:
                current = test
            else:
                if current:
                    lines.append(current)
                current = word
        if current:
            lines.append(current)
        
        # Limit to max lines
        lines = lines[-self._max_lines:]
        
        # Draw background
        line_h = self._font.get_linesize()
        total_h = len(lines) * line_h + 2 * self._padding
        bg_rect = pygame.Rect(0, self._screen_h - total_h, self._screen_w, total_h)
        bg_surface = pygame.Surface((bg_rect.w, bg_rect.h), pygame.SRCALPHA)
        bg_surface.fill(self._bg_color)
        surface.blit(bg_surface, bg_rect)
        
        # Draw text
        y = self._screen_h - total_h + self._padding
        for line in lines:
            text_surf = self._font.render(line, True, self._text_color)
            x = (self._screen_w - text_surf.get_width()) // 2
            surface.blit(text_surf, (x, y))
            y += line_h
```

# server/shot_events.py
"""Generalized Shot Event system for party ceremonies and toasts."""
from dataclasses import dataclass, field
from typing import Optional

DEBUG_SHOT_EVENTS = True

@dataclass
class ShotEvent:
    name: str
    tone: str  # "solemn", "celebratory", "fun"
    trigger_type: str  # "auto", "voice", "admin"
    voice_keywords: list[str] = field(default_factory=list)
    phases: list[str] = field(default_factory=lambda: ["announcement", "countdown", "toast", "recovery"])
    announcement_text: str = ""
    silence_text: str = ""  # only for solemn events
    toast_text: str = ""
    recovery_line: str = ""
    countdown: bool = True
    music_file: Optional[str] = None
    music_duration: int = 0
    skip_key: Optional[str] = None  # e.g., "ctrl+shift+l"
    fired: bool = False

class ShotEventManager:
    def __init__(self):
        self.events: dict[str, ShotEvent] = {}
        self._active_event: Optional[str] = None
        if DEBUG_SHOT_EVENTS:
            print("[DEBUG_SHOT_EVENTS] ShotEventManager: initialized")
    
    def register(self, event: ShotEvent):
        self.events[event.name] = event
        if DEBUG_SHOT_EVENTS:
            print(f"[DEBUG_SHOT_EVENTS] register: {event.name} ({event.tone})")
    
    def trigger(self, name: str) -> dict:
        if name not in self.events:
            return {"status": "not_found", "event": name}
        event = self.events[name]
        if event.fired:
            return {"status": "already_fired", "event": name}
        event.fired = True
        self._active_event = name
        if DEBUG_SHOT_EVENTS:
            print(f"[DEBUG_SHOT_EVENTS] trigger: {name}")
        return {"status": "triggered", "event": name}
    
    def complete(self, name: str):
        if name in self.events:
            self._active_event = None
            if DEBUG_SHOT_EVENTS:
                print(f"[DEBUG_SHOT_EVENTS] complete: {name}")
    
    def reset(self, name: str):
        if name in self.events:
            self.events[name].fired = False
            if DEBUG_SHOT_EVENTS:
                print(f"[DEBUG_SHOT_EVENTS] reset: {name}")
    
    def check_voice_trigger(self, text: str) -> Optional[ShotEvent]:
        lower = text.lower()
        for event in self.events.values():
            if event.fired or event.trigger_type not in ("voice", "auto"):
                continue
            for kw in event.voice_keywords:
                if kw.lower() in lower:
                    if DEBUG_SHOT_EVENTS:
                        print(f"[DEBUG_SHOT_EVENTS] voice_trigger: '{kw}' matched for {event.name}")
                    return event
        return None
    
    def list_events(self) -> list[dict]:
        return [{"name": e.name, "fired": e.fired, "tone": e.tone}
                for e in self.events.values()]
    
    def get_countdown_texts(self) -> list[str]:
        return [
            "TEN-a!", "NINE-a!", "EIGHT-a!", "SEVEN-a!", "SIX-a!",
            "FIVE-a!", "FOUR-a!", "THREE-a!", "TWO-a!", "ONE-a!"
        ]
    
    async def precache_countdown_audio(self, tts_func):
        """Pre-cache TTS audio for all countdown numbers at startup.
        Call this during server init to guarantee zero-latency countdown."""
        self._countdown_cache = {}
        for text in self.get_countdown_texts():
            audio = await tts_func(text)
            self._countdown_cache[text] = audio
            if DEBUG_SHOT_EVENTS:
                print(f"[DEBUG_SHOT_EVENTS] precached countdown: {text}")
    
    def get_cached_countdown(self, text: str):
        """Return pre-cached audio for a countdown number, or None."""
        return self._countdown_cache.get(text)
    
    @property
    def is_active(self) -> bool:
        return self._active_event is not None
    
    @property
    def active_event(self) -> Optional[ShotEvent]:
        if self._active_event:
            return self.events.get(self._active_event)
        return None
```

async def _run_shot_event(event: ShotEvent):
    """Execute a shot event's phases sequentially."""
    state_current["memorial_active"] = True  # reuse existing flag
    
    for phase in event.phases:
        if phase == "announcement":
            text = event.announcement_text
            audio = await synthesize_tts(text)
            await broadcast_event("memorial_event", {"phase": "announcement", "text": text}, audio)
            await asyncio.sleep(len(audio) / 48000 + 2)
        
        elif phase == "silence":
            text = event.silence_text
            audio = await synthesize_tts(text)
            await broadcast_event("memorial_event", {"phase": "silence", "text": text}, audio)
            await asyncio.sleep(len(audio) / 48000 + 5)
        
        elif phase == "countdown":
            countdown_texts = shot_event_manager.get_countdown_texts()
            for ct in countdown_texts:
                # Use pre-cached audio for zero-latency countdown
                audio = shot_event_manager.get_cached_countdown(ct)
                if audio is None:
                    audio = await synthesize_tts(ct)  # fallback
                await broadcast_event("memorial_event", {"phase": "countdown", "text": ct, "overlay": True}, audio)
                await asyncio.sleep(1.0)
        
        elif phase == "toast":
            text = event.toast_text
            audio = await synthesize_tts(text)
            await broadcast_event("memorial_event", {"phase": "toast", "text": text}, audio)
            await asyncio.sleep(len(audio) / 48000 + 2)
        
        elif phase == "music":
            if event.music_file:
                await broadcast_event("memorial_event", {"phase": "music", "file": event.music_file})
                await asyncio.sleep(event.music_duration)
        
        elif phase == "recovery":
            text = event.recovery_line
            audio = await synthesize_tts(text)
            await broadcast_event("memorial_event", {"phase": "fadeout", "text": text}, audio)
            await asyncio.sleep(len(audio) / 48000 + 5)
    
    shot_event_manager.complete(event.name)
    await asyncio.sleep(15)
    state_current["memorial_active"] = False

```python
import random

class EasterEggScheduler:
    """Schedules the 'N-Word Incident' easter egg 3-5 times per party."""
    
    EASTER_EGG_TEXT = "Jacob Hoppenstedt said the N-Word earlier at this party. Oops! I shouldn't have-a said that!"
    
    def __init__(self, party_duration_hours=8):
        self._fire_count = 0
        num_fires = random.randint(3, 5)
        # Generate firing times: random slots with 30-min minimum gap
        total_seconds = party_duration_hours * 3600
        slot_size = 1800  # 30 minutes
        num_slots = total_seconds // slot_size
        slots = sorted(random.sample(range(num_slots), min(num_fires, num_slots)))
        base_time = time.time()
        self.firing_times = [base_time + s * slot_size + random.randint(0, slot_size - 1) for s in slots]
    
    def should_fire(self) -> bool:
        now = time.time()
        for ft in self.firing_times:
            if ft <= now and self._fire_count < len(self.firing_times):
                return True
        return False
    
    def record_fired(self):
        self._fire_count += 1
        # Remove past firing times
        now = time.time()
        self.firing_times = [t for t in self.firing_times if t > now]
    
    def get_text(self) -> str:
        return self.EASTER_EGG_TEXT
```
